cmake_minimum_required(VERSION 3.16)
project(atlas_rest VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(atlas-server src/main.cpp)
target_compile_options(atlas-server PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
target_link_libraries(atlas-server PRIVATE Threads::Threads)
include(CTest)
if(BUILD_TESTING)
  add_executable(atlas-core-tests tests/core.cpp)
  target_link_libraries(atlas-core-tests PRIVATE Threads::Threads)
  add_test(NAME core COMMAND atlas-core-tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:atlas-server>)
endif()
install(TARGETS atlas-server RUNTIME DESTINATION bin)
